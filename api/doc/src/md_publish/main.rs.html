37fn main() -> Result<()> {
38    let args: Vec<String> = std::env::args().collect();
39    
40    if args.len() > 1 && args[1] == "supports" {
41        return Ok(());
42    }
43
44    if args.len() > 1 && args[1] == "doctor" {
45        return run_doctor();
46    }
47
48    // Load config from book.toml
49    let config_content = std::fs::read_to_string("book.toml").unwrap_or_default();
50    let book_config: BookConfig = toml::from_str(&config_content).unwrap_or(BookConfig { preprocessor: None });
51    let ingest_config = book_config.preprocessor.and_then(|p| p.ingest).unwrap_or(IngestConfig {
52        downloads_path: Some("/mnt/c/Users/ashut/Downloads".to_string()),
53        text_source: None,
54        image_source: None,
55        video_source: None,
56        lightning_address: Some("shutosha@primal.net".to_string()),
57        title_word_limit: Some(5),
58        recent_window_size: Some(21),
59        podcast_html: None,
60        visual_html: None,
61    });
62
63    let mut number = None;
64    let mut do_text = false;
65    let mut do_image = false;
66    let mut do_video = false;
67    let mut do_park = None;
68    let mut do_unpark = None;
69    let mut do_list_parked = false;
70    let mut title_override = None;
71    let mut source = ingest_config.downloads_path.clone().unwrap_or_else(|| "/mnt/c/Users/ashut/Downloads".to_string());
72
73    let mut i = 1;
74    while i < args.len() {
75        match args[i].as_str() {
76            "--text" => do_text = true,
77            "--image" => do_image = true,
78            "--video" => do_video = true,
79            "--park" => {
80                if i + 1 < args.len() {
81                    do_park = Some(args[i+1].clone());
82                    i += 1;
83                }
84            },
85            "--unpark" => {
86                if i + 2 < args.len() {
87                    do_unpark = Some((args[i+1].clone(), args[i+2].clone()));
88                    i += 2;
89                }
90            },
91            "--list-parked" => do_list_parked = true,
92            "-t" | "--title" => {
93                if i + 1 < args.len() {
94                    title_override = Some(args[i+1].as_str());
95                    i += 1;
96                }
97            },
98            "-s" | "--source" => {
99                if i + 1 < args.len() {
100                    source = args[i+1].to_string();
101                    i += 1;
102                }
103            },
104            num if num.chars().all(|c| c.is_digit(10)) => number = Some(num.to_string()),
105            _ => {}
106        }
107        i += 1;
108    }
109
110    if do_list_parked {
111        return list_parked();
112    }
113
114    if let Some(num) = do_park {
115        return park_episode(&num, &ingest_config);
116    }
117
118    if let Some((old_num, new_num)) = do_unpark {
119        return unpark_episode(&old_num, &new_num, &ingest_config);
120    }
121
122    if let Some(num) = number {
123        if do_text { 
124            let text_src = ingest_config.text_source.clone().unwrap_or(source.clone());
125            ingest_text(&num, &text_src, title_override, &ingest_config)?; 
126        }
127        if do_image { 
128            let image_src = ingest_config.image_source.clone().unwrap_or(source.clone());
129            ingest_image(&num, &image_src, &ingest_config)?; 
130        }
131        if do_video { 
132            let video_src = ingest_config.video_source.clone().unwrap_or(source.clone());
133            ingest_video(&num, &video_src, &ingest_config)?; 
134        }
135    } else {
136        // Standard preprocessor mode
137        let mut input = String::new();
138        std::io::Read::read_to_string(&mut std::io::stdin(), &mut input)?;
139        
140        // Use the mdbook_preprocessor crate to parse correctly for 0.5.x
141        let (_ctx, book) = mdbook_preprocessor::parse_input(input.as_bytes())?;
142        
143        // Serialize and output the Book object
144        let output_json = serde_json::to_string(&book)?;
145        print!("{}", output_json);
146    }
147
148    Ok(())
149}

151fn ingest_text(number: &str, source: &str, title: Option<&str>, config: &IngestConfig) -> Result<()> {
152    eprintln!("📖 Ingesting text for episode {}...", number);
153    let mut files: Vec<PathBuf> = Vec::new();
154    
155    // Support .json, .rs, and .md
156    for ext in &["json", "rs", "md"] {
157        let pattern = format!("{}/*.{}", source, ext);
158        if let Ok(paths) = glob(&pattern) {
159            for path in paths.filter_map(Result::ok) {
160                let filename = path.file_name().unwrap().to_str().unwrap();
161                if !["SUMMARY.md", "cover.md"].contains(&filename) {
162                    files.push(path);
163                }
164            }
165        }
166    }
167    
168    files.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
169
170    if let Some(path) = files.first() {
171        eprintln!("📄 Found source: {}", path.display());
172        let content = std::fs::read_to_string(path)?;
173        let hardened = sanitizer::process_content(content, number, title, config.title_word_limit.unwrap_or(5));
174        std::fs::write(format!("src/{}.md", number), hardened)?;
175        eprintln!("✅ Ingested text to src/{}.md", number);
176        
177        // Sync SUMMARY.md
178        if let Err(e) = update_summary(config) {
179            eprintln!("⚠️ Failed to update SUMMARY.md: {}", e);
180        } else {
181            eprintln!("✅ Synchronized SUMMARY.md");
182        }
183    }
184    Ok(())
185}

195fn update_summary(config: &IngestConfig) -> Result<()> {
196    let summary_path = "src/SUMMARY.md";
197    let window_size = config.recent_window_size.unwrap_or(21);
198    
199    let mut all_files = Vec::new();
200    let pattern = "src/*.md";
201    for entry in glob(pattern)? {
202        if let Ok(path) = entry {
203            let filename = path.file_name().unwrap().to_str().unwrap();
204            if ["SUMMARY.md", "cover.md"].contains(&filename) { continue; }
205            let base = filename.trim_end_matches(".md").trim();
206            let md_content = std::fs::read_to_string(&path)?;
207            let h1_regex = regex::Regex::new(r"(?m)^#\s+(?:(?:\d+)\s*[:\s]*)?\s*(.*)$").unwrap();
208            let title = if let Some(caps) = h1_regex.captures(&md_content) {
209                caps.get(1).unwrap().as_str().trim().to_string()
210            } else { "Untitled".to_string() };
211            let mtime = std::fs::metadata(&path)?.modified()?;
212            let number = base.parse::<u32>().ok();
213            all_files.push(EpisodeEntry { number, filename: base.to_string(), title, mtime });
214        }
215    }
216
217    let mut numbered_active: Vec<_> = all_files.iter().filter(|e| e.number.is_some() && !e.filename.starts_with("_")).cloned().collect();
218    let mut wip_parked: Vec<_> = all_files.iter().filter(|e| e.filename.starts_with("_")).cloned().collect();
219
220    numbered_active.sort_by(|a, b| b.number.unwrap().cmp(&a.number.unwrap()));
221    wip_parked.sort_by(|a, b| b.mtime.cmp(&a.mtime));
222
223    let recents: Vec<_> = numbered_active.iter().take(window_size).collect();
224    let overflow_numbered: Vec<_> = numbered_active.iter().skip(window_size).cloned().collect();
225
226    eprintln!("📊 Indexer: {} Recents, {} WIP", recents.len(), wip_parked.len());
227    
228    let original_content = std::fs::read_to_string(summary_path)?;
229    let mut final_lines = Vec::new();
230    
231    for line in original_content.lines() {
232        if line.contains("# Recent") || line.contains("<!-- RECENT_START -->") { break; }
233        final_lines.push(line.to_string());
234    }
235
236    final_lines.push("\n# Recent ..".to_string());
237    final_lines.push("<!-- RECENT_START -->".to_string());
238    for ep in recents { final_lines.push(format!("- [{} : {}]({}.md)", ep.number.unwrap(), ep.title, ep.filename)); }
239    final_lines.push("<!-- RECENT_END -->".to_string());
240
241    final_lines.push("\n# WIP / Parked".to_string());
242    if wip_parked.is_empty() {
243        final_lines.push("- [None at this moment. Join us on GitHub!](github.md)".to_string());
244    } else {
245        for ep in wip_parked {
246            let display_num = ep.filename.trim_start_matches('_');
247            final_lines.push(format!("- [{} : {}]({}.md)", display_num, ep.title, ep.filename));
248        }
249    }
250
251    final_lines.push("\n# Archive".to_string());
252    if !overflow_numbered.is_empty() {
253        final_lines.push("## Older Episodes".to_string());
254        for ep in overflow_numbered {
255            final_lines.push(format!("- [{} : {}]({}.md)", ep.number.unwrap(), ep.title, ep.filename));
256        }
257    }
258
259    let mut in_thematic_zone = false;
260    let mut thematic_buffer = Vec::new();
261    let num_regex = regex::Regex::new(r"\d+\.md").unwrap();
262    for line in original_content.lines() {
263        if line.contains("<!-- RECENT_END -->") { in_thematic_zone = true; continue; }
264        if in_thematic_zone {
265            if line.contains("# Recent") || line.contains("# WIP") || line.contains("# Archive") || line.contains("# Repository") { continue; }
266            if num_regex.is_match(line) { continue; }
267            if thematic_buffer.is_empty() && line.trim().is_empty() { continue; }
268            thematic_buffer.push(line.to_string());
269        }
270    }
271    final_lines.extend(thematic_buffer);
272    std::fs::write(summary_path, final_lines.join("\n"))?;
273    Ok(())
274}

276fn ingest_image(number: &str, source: &str, _config: &IngestConfig) -> Result<()> {
277    eprintln!("🎨 Ingesting image for episode {}...", number);
278    let img_dir = "src/img";
279    std::fs::create_dir_all(img_dir)?;
280    let mut images: Vec<PathBuf> = glob(&format!("{}/*{}*.png", source, number))?.filter_map(Result::ok).collect();
281    images.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
282
283    if let Some(path) = images.first() {
284        let dest = format!("{}/{}.png", img_dir, number);
285        // Avoid truncating if copying from within the same folder (redundant check but safe)
286        if path.as_path() != std::path::Path::new(&dest) {
287            std::fs::copy(path, &dest)?;
288        }
289        eprintln!("✅ Ingested cover art to {} (archived, no Markdown injection)", dest);
290    }
291    Ok(())
292}

299
300    // 1. Migration from Downloads
301    let pattern = format!("{}/*{}*.mp4", source, number);
302    if let Ok(paths) = glob(&pattern) {
303        for path in paths.filter_map(Result::ok) {
304            let filename = path.file_name().unwrap().to_str().unwrap();
305            let dest = format!("{}/{}", vid_dir, filename);
306            std::fs::copy(&path, &dest)?;
307            eprintln!("✅ Migrated: {}", filename);
308        }
309    }
310
311    // 2. Discover LOCAL videos for this episode only
312    let mut local_vids = Vec::new();
313    if let Ok(paths) = glob(&format!("{}/{}*.mp4", vid_dir, number)) {
314        for path in paths.filter_map(Result::ok) {
315            local_vids.push(path);
316        }
317    }
318
319    // Intelligent Sequence Sorting (Bucket Logic):
320    // Priority 0: Base episode or Intro (241.mp4, 241-Intro.mp4)
321    // Priority 1: All indexed files sorted as strings (241-2, 241-21, 241-3)
322    local_vids.sort_by(|a, b| {
323        let get_prio_and_key = |path: &std::path::PathBuf| -> (i32, String) {
324            let filename = path.file_name().unwrap().to_str().unwrap().to_lowercase();

327            let suffix = if base.starts_with(number) {

333            if suffix.is_empty() || suffix == "-intro" || suffix == "-0" {
334                (0, filename)
335            } else {
336                // Return Priority 1 and the suffix itself for string-based bucket sorting
337                (1, filename)
338            }
339        };
340
341        let val_a = get_prio_and_key(a);
342        let val_b = get_prio_and_key(b);
343        val_a.cmp(&val_b)

473    eprintln!("🅿️  Currently Parked Episodes:");

479            let h1_regex = regex::Regex::new(r"(?m)^#\s+(?:(?:\d+)\s*[:\s]*)?\s*(.*)$").unwrap();

487fn park_episode(number: &str, config: &IngestConfig) -> Result<()> {
488    let videos: Vec<_> = glob(&format!("src/vid/{}*.mp4", number))?.filter_map(Result::ok).collect();
489    if !videos.is_empty() {
490        eprintln!("❌ Cannot park episode {}: It has infographic videos and is 'cast in stone'.", number);
491        anyhow::bail!("Episode is immutable");
492    }
493    let src = format!("src/{}.md", number);
494    let dest = format!("src/_{}.md", number);
495    if std::path::Path::new(&src).exists() {
496        std::fs::rename(&src, &dest)?;
497        eprintln!("✅ Parked episode {} -> {}", src, dest);
498        update_summary(config)?; 
499    } else {
500        eprintln!("⚠️ Episode {} not found in src/", number);
501    }
502    Ok(())
503}

505fn unpark_episode(old_number: &str, new_number: &str, config: &IngestConfig) -> Result<()> {
506    let src = format!("src/_{}.md", old_number);
507    let dest = format!("src/{}.md", new_number);
508    if !std::path::Path::new(&src).exists() { anyhow::bail!("Source not found"); }
509    std::fs::rename(&src, &dest)?;
510    let mut content = std::fs::read_to_string(&dest)?;
511    content = content.replace(&format!("# {}", old_number), &format!("# {}", new_number));
512    std::fs::write(&dest, content)?;
