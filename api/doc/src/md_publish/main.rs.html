46fn main() -> Result<()> {
47    let args: Vec<String> = std::env::args().collect();
48    if args.len() > 1 && args[1] == "supports" { return Ok(()); }
49    if args.len() > 1 && args[1] == "doctor" { return run_doctor(); }
50
51    let config_content = std::fs::read_to_string("book.toml").unwrap_or_default();
52    let book_config: BookConfig = toml::from_str(&config_content).unwrap_or(BookConfig { preprocessor: None });
53    let ingest_config = book_config.preprocessor.and_then(|p| p.ingest).unwrap_or(IngestConfig {
54        downloads_path: Some("/mnt/c/Users/ashut/Downloads".to_string()),
55        text_source: None,
56        image_source: None,
57        video_source: None,
58        lightning_address: Some("shutosha@primal.net".to_string()),
59        title_word_limit: Some(5),
60        recent_window_size: Some(21),
61        podcast_html: None,
62        visual_html: None,
63    });
64
65    let mut number = None;
66    let mut do_text = false;
67    let mut do_image = false;
68    let mut do_video = false;
69    let mut do_park = None;
70    let mut do_unpark = None;
71    let mut do_list_parked = false;
72    let mut title_override = None;
73    let mut source = ingest_config.downloads_path.clone().unwrap_or_else(|| "/mnt/c/Users/ashut/Downloads".to_string());
74
75    let mut i = 1;
76    while i < args.len() {
77        match args[i].as_str() {
78            "--text" => do_text = true,
79            "--image" => do_image = true,
80            "--video" => do_video = true,
81            "--park" => { if i + 1 < args.len() { do_park = Some(args[i+1].clone()); i += 1; } },
82            "--unpark" => { if i + 2 < args.len() { do_unpark = Some((args[i+1].clone(), args[i+2].clone())); i += 2; } },
83            "--list-parked" => do_list_parked = true,
84            "-t" | "--title" => { if i + 1 < args.len() { title_override = Some(args[i+1].as_str()); i += 1; } },
85            "-s" | "--source" => { if i + 1 < args.len() { source = args[i+1].to_string(); i += 1; } },
86            num if num.chars().all(|c| c.is_digit(10)) => number = Some(num.to_string()),
87            _ => {}
88        }
89        i += 1;
90    }
91
92    if do_list_parked { return list_parked(); }
93    if let Some(num) = do_park { return park_episode(&num, &ingest_config); }
94    if let Some((old_num, new_num)) = do_unpark { return unpark_episode(&old_num, &new_num, &ingest_config); }
95
96    if let Some(num) = number {
97        if do_text { 
98            let text_src = ingest_config.text_source.clone().unwrap_or(source.clone());
99            ingest_text(&num, &text_src, title_override, &ingest_config)?; 
100        }
101        if do_image { 
102            let image_src = ingest_config.image_source.clone().unwrap_or(source.clone());
103            ingest_image(&num, &image_src, &ingest_config)?; 
104        }
105        if do_video { 
106            let video_src = ingest_config.video_source.clone().unwrap_or(source.clone());
107            ingest_video(&num, &video_src, &ingest_config)?; 
108        }
109        update_summary(&ingest_config)?;
110    } else {
111        let mut input = String::new();
112        std::io::Read::read_to_string(&mut std::io::stdin(), &mut input)?;
113        let (_ctx, book) = mdbook_preprocessor::parse_input(input.as_bytes())?;
114        let output_json = serde_json::to_string(&book)?;
115        print!("{}", output_json);
116    }
117    Ok(())
118}

120fn ingest_text(number: &str, source: &str, title: Option<&str>, config: &IngestConfig) -> Result<()> {
121    eprintln!("📖 Ingesting text for episode {}...", number);
122    let mut files: Vec<PathBuf> = Vec::new();
123    for ext in &["json", "rs", "md"] {
124        let pattern = format!("{}/*.{}", source, ext);
125        if let Ok(paths) = glob(&pattern) {
126            for path in paths.filter_map(Result::ok) {
127                let filename = path.file_name().unwrap().to_str().unwrap();
128                if !["SUMMARY.md", "cover.md"].contains(&filename) { files.push(path); }
129            }
130        }
131    }
132    files.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
133    if let Some(path) = files.first() {
134        eprintln!("📄 Found source: {}", path.display());
135        let content = std::fs::read_to_string(path)?;
136        let hardened = sanitizer::process_content(content, number, title, config.title_word_limit.unwrap_or(5));
137        std::fs::write(format!("src/{}.md", number), hardened)?;
138        eprintln!("✅ Ingested text to src/{}.md", number);
139    }
140    Ok(())
141}

283fn ingest_image(number: &str, source: &str, _config: &IngestConfig) -> Result<()> {
284    eprintln!("🎨 Ingesting image for episode {}...", number);
285    let img_dir = "src/img";
286    std::fs::create_dir_all(img_dir)?;
287    let mut images: Vec<PathBuf> = glob(&format!("{}/*{}*.png", source, number))?.filter_map(Result::ok).collect();
288    images.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
289    if let Some(path) = images.first() {
290        let dest = format!("{}/{}.png", img_dir, number);
291        if path.as_path() != std::path::Path::new(&dest) { std::fs::copy(path, &dest)?; }
292    }
293    Ok(())
294}

315fn list_parked() -> Result<()> {
316    eprintln!("🅿️  Currently Parked Episodes:");
317    for entry in glob("src/_[0-9]*.md")? {
318        if let Ok(path) = entry {
319            let filename = path.file_name().unwrap().to_str().unwrap();
320            let base = filename.trim_end_matches(".md");
321            let content = std::fs::read_to_string(&path)?;
322            let h1_regex = regex::Regex::new(r"(?m)^#\s+(?:(?:\d+)\s*[:\s]*)?\s*(.*)$").unwrap();
323            let title = if let Some(caps) = h1_regex.captures(&content) { caps.get(1).unwrap().as_str().trim() } else { "Untitled" };
324            println!("  [{}] : {}", &base[1..], title);
325        }
326    }
327    Ok(())

330fn park_episode(number: &str, config: &IngestConfig) -> Result<()> {
331    let videos: Vec<_> = glob(&format!("src/vid/{}*.mp4", number))?.filter_map(Result::ok).collect();
332    if !videos.is_empty() { anyhow::bail!("Episode is immutable"); }
333    let src = format!("src/{}.md", number);
334    if std::path::Path::new(&src).exists() {
335        std::fs::rename(&src, format!("src/_{}.md", number))?;
336        update_summary(config)?; 
337    }
338    Ok(())
339}

341fn unpark_episode(old_number: &str, new_number: &str, config: &IngestConfig) -> Result<()> {
342    let src = format!("src/_{}.md", old_number);
343    let dest = format!("src/{}.md", new_number);
344    if !std::path::Path::new(&src).exists() { anyhow::bail!("Source not found"); }
345    std::fs::rename(&src, &dest)?;
346    let mut content = std::fs::read_to_string(&dest)?;
347    content = content.replace(&format!("# {}", old_number), &format!("# {}", new_number));
348    std::fs::write(&dest, content)?;
349    update_summary(config)?;
350    Ok(())
