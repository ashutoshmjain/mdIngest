36fn main() -> Result<()> {
37    let args: Vec<String> = std::env::args().collect();
38    
39    if args.len() > 1 && args[1] == "supports" {
40        return Ok(());
41    }
42
43    if args.len() > 1 && args[1] == "doctor" {
44        return run_doctor();
45    }
46
47    // Load config from book.toml
48    let config_content = std::fs::read_to_string("book.toml").unwrap_or_default();
49    let book_config: BookConfig = toml::from_str(&config_content).unwrap_or(BookConfig { preprocessor: None });
50    let ingest_config = book_config.preprocessor.and_then(|p| p.ingest).unwrap_or(IngestConfig {
51        downloads_path: Some("/mnt/c/Users/ashut/Downloads".to_string()),
52        text_source: None,
53        image_source: None,
54        video_source: None,
55        lightning_address: Some("shutosha@primal.net".to_string()),
56        title_word_limit: Some(5),
57        podcast_html: None,
58        visual_html: None,
59    });
60
61    let mut number = None;
62    let mut do_text = false;
63    let mut do_image = false;
64    let mut do_video = false;
65    let mut title_override = None;
66    let mut source = ingest_config.downloads_path.clone().unwrap_or_else(|| "/mnt/c/Users/ashut/Downloads".to_string());
67
68    let mut i = 1;
69    while i < args.len() {
70        match args[i].as_str() {
71            "--text" => do_text = true,
72            "--image" => do_image = true,
73            "--video" => do_video = true,
74            "-t" | "--title" => {
75                if i + 1 < args.len() {
76                    title_override = Some(args[i+1].as_str());
77                    i += 1;
78                }
79            },
80            "-s" | "--source" => {
81                if i + 1 < args.len() {
82                    source = args[i+1].to_string();
83                    i += 1;
84                }
85            },
86            num if num.chars().all(|c| c.is_digit(10)) => number = Some(num.to_string()),
87            _ => {}
88        }
89        i += 1;
90    }
91
92    if let Some(num) = number {
93        if do_text { 
94            let text_src = ingest_config.text_source.clone().unwrap_or(source.clone());
95            ingest_text(&num, &text_src, title_override, &ingest_config)?; 
96        }
97        if do_image { 
98            let image_src = ingest_config.image_source.clone().unwrap_or(source.clone());
99            ingest_image(&num, &image_src, &ingest_config)?; 
100        }
101        if do_video { 
102            let video_src = ingest_config.video_source.clone().unwrap_or(source.clone());
103            ingest_video(&num, &video_src, &ingest_config)?; 
104        }
105    } else {
106        if args.len() > 1 && !args[1].starts_with('-') {
107            // Support positional number
108        } else {
109            // Standard preprocessor mode
110            let (_ctx, book) = mdbook::preprocess::CmdPreprocessor::parse_input(std::io::stdin())?;
111            print!("{}", serde_json::to_string(&book)?);
112        }
113    }
114
115    Ok(())
116}

118fn ingest_text(number: &str, source: &str, title: Option<&str>, config: &IngestConfig) -> Result<()> {
119    eprintln!("📖 Ingesting text for episode {}...", number);
120    let mut files: Vec<PathBuf> = Vec::new();
121    
122    // Support .json, .rs, and .md
123    for ext in &["json", "rs", "md"] {
124        let pattern = format!("{}/*.{}", source, ext);
125        if let Ok(paths) = glob(&pattern) {
126            for path in paths.filter_map(Result::ok) {
127                let filename = path.file_name().unwrap().to_str().unwrap();
128                if !["SUMMARY.md", "cover.md"].contains(&filename) {
129                    files.push(path);
130                }
131            }
132        }
133    }
134    
135    files.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
136
137    if let Some(path) = files.first() {
138        eprintln!("📄 Found source: {}", path.display());
139        let content = std::fs::read_to_string(path)?;
140        let hardened = sanitizer::process_content(content, number, title, config.title_word_limit.unwrap_or(5));
141        std::fs::write(format!("src/{}.md", number), hardened)?;
142        eprintln!("✅ Ingested text to src/{}.md", number);
143        
144        // Sync SUMMARY.md
145        if let Err(e) = update_summary(number) {
146            eprintln!("⚠️ Failed to update SUMMARY.md: {}", e);
147        } else {
148            eprintln!("✅ Synchronized SUMMARY.md");
149        }
150    }
151    Ok(())
152}

154fn update_summary(number: &str) -> Result<()> {
155    let summary_path = "src/SUMMARY.md";
156    let content = std::fs::read_to_string(summary_path)?;
157    let md_path = format!("src/{}.md", number);
158    
159    // Extract title from the newly created markdown file
160    let md_content = std::fs::read_to_string(&md_path)?;
161    let h1_regex = regex::Regex::new(r"(?m)^#\s+\d+\s*:\s*(.*)$").unwrap();
162    let title = if let Some(caps) = h1_regex.captures(&md_content) {
163        caps.get(1).unwrap().as_str().trim()
164    } else {
165        "Untitled"
166    };
167
168    let new_entry = format!("- [{} : {}]({}.md)", number, title, number);
169    
170    // Check if already in SUMMARY.md
171    if content.contains(&format!("({}.md)", number)) {
172        return Ok(());
173    }
174
175    // Insert after <!-- RECENT_START -->
176    let mut lines: Vec<String> = content.lines().map(|s| s.to_string()).collect();
177    if let Some(pos) = lines.iter().position(|l| l.contains("<!-- RECENT_START -->")) {
178        lines.insert(pos + 1, new_entry);
179        std::fs::write(summary_path, lines.join("\n"))?;
180    }
181
182    Ok(())
183}

185fn ingest_image(number: &str, source: &str, _config: &IngestConfig) -> Result<()> {
186    eprintln!("🎨 Ingesting image for episode {}...", number);
187    let img_dir = "src/img";
188    std::fs::create_dir_all(img_dir)?;
189    let mut images: Vec<PathBuf> = glob(&format!("{}/*{}*.png", source, number))?.filter_map(Result::ok).collect();
190    images.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
191
192    if let Some(path) = images.first() {
193        let dest = format!("{}/{}.png", img_dir, number);
194        // Avoid truncating if copying from within the same folder (redundant check but safe)
195        if path.as_path() != std::path::Path::new(&dest) {
196            std::fs::copy(path, &dest)?;
197        }
198        eprintln!("✅ Ingested cover art to {} (archived, no Markdown injection)", dest);
199    }
200    Ok(())
201}

203fn ingest_video(number: &str, source: &str, _config: &IngestConfig) -> Result<()> {
204    eprintln!("🎬 Ingesting video for episode {}...", number);
205    
206    let vid_dir = "src/vid";
207    std::fs::create_dir_all(vid_dir)?;
208
209    // 1. Migration from Downloads
210    let pattern = format!("{}/*{}*.mp4", source, number);
211    if let Ok(paths) = glob(&pattern) {
212        for path in paths.filter_map(Result::ok) {
213            let filename = path.file_name().unwrap().to_str().unwrap();
214            let dest = format!("{}/{}", vid_dir, filename);
215            std::fs::copy(&path, &dest)?;
216            eprintln!("✅ Migrated: {}", filename);
217        }
218    }
219
220    // 2. Discover LOCAL videos for this episode only
221    let mut local_vids = Vec::new();
222    if let Ok(paths) = glob(&format!("{}/{}*.mp4", vid_dir, number)) {
223        for path in paths.filter_map(Result::ok) {
224            local_vids.push(path);
225        }
226    }
227
228    // Intelligent Sequence Sorting (Bucket Logic):
229    // Priority 0: Base episode or Intro (241.mp4, 241-Intro.mp4)
230    // Priority 1: All indexed files sorted as strings (241-2, 241-21, 241-3)
231    local_vids.sort_by(|a, b| {
232        let get_prio_and_key = |path: &std::path::PathBuf| -> (i32, String) {
233            let filename = path.file_name().unwrap().to_str().unwrap().to_lowercase();
234            let base = filename.strip_suffix(".mp4").unwrap_or(&filename);
235            
236            let suffix = if base.starts_with(number) {
237                &base[number.len()..]
238            } else {
239                base
240            };
241
242            if suffix.is_empty() || suffix == "-intro" || suffix == "-0" {
243                (0, filename)
244            } else {
245                // Return Priority 1 and the suffix itself for string-based bucket sorting
246                (1, filename)
247            }
248        };
249
250        let val_a = get_prio_and_key(a);
251        let val_b = get_prio_and_key(b);
252        val_a.cmp(&val_b)
253    });
254
255    if local_vids.is_empty() {
256        eprintln!("⚠️ No videos found for episode {}", number);

369fn run_doctor() -> Result<()> {
370    match Command::new("mdbook").arg("--version").output() {
371        Ok(out) => eprintln!("✅ mdbook: {}", String::from_utf8_lossy(&out.stdout).trim()),
372        Err(_) => eprintln!("❌ mdbook: Not found"),
373    }
374    match Command::new("mdbook-katex").arg("--version").output() {
375        Ok(out) => eprintln!("✅ mdbook-katex: {}", String::from_utf8_lossy(&out.stdout).trim()),
376        Err(_) => eprintln!("❌ mdbook-katex: Not found"),
377    }
378    Ok(())
