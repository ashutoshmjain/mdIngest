47fn main() -> Result<()> {
48    let args: Vec<String> = std::env::args().collect();
49    if args.len() > 1 && args[1] == "supports" { return Ok(()); }
50    if args.len() > 1 && args[1] == "doctor" { return run_doctor(); }
51
52    let config_content = std::fs::read_to_string("book.toml").unwrap_or_default();
53    let book_config: BookConfig = toml::from_str(&config_content).unwrap_or(BookConfig { wip_threshold: Some(240), preprocessor: None });
54    let ingest_config = book_config.preprocessor.and_then(|p| p.ingest).unwrap_or(IngestConfig {
55        downloads_path: Some("/mnt/c/Users/ashut/Downloads".to_string()),
56        text_source: None,
57        image_source: None,
58        video_source: None,
59        lightning_address: Some("shutosha@primal.net".to_string()),
60        title_word_limit: Some(5),
61        recent_window_size: Some(21),
62        podcast_html: None,
63        visual_html: None,
64    });
65
66    let mut number = None;
67    let mut do_text = false;
68    let mut do_image = false;
69    let mut do_video = false;
70    let mut do_park = None;
71    let mut do_unpark = None;
72    let mut do_list_parked = false;
73    let mut title_override = None;
74    let mut source = ingest_config.downloads_path.clone().unwrap_or_else(|| "/mnt/c/Users/ashut/Downloads".to_string());
75
76    let mut i = 1;
77    while i < args.len() {
78        match args[i].as_str() {
79            "--text" => do_text = true,
80            "--image" => do_image = true,
81            "--video" => do_video = true,
82            "--park" => { if i + 1 < args.len() { do_park = Some(args[i+1].clone()); i += 1; } },
83            "--unpark" => { if i + 2 < args.len() { do_unpark = Some((args[i+1].clone(), args[i+2].clone())); i += 2; } },
84            "--list-parked" => do_list_parked = true,
85            "-t" | "--title" => { if i + 1 < args.len() { title_override = Some(args[i+1].as_str()); i += 1; } },
86            "-s" | "--source" => { if i + 1 < args.len() { source = args[i+1].to_string(); i += 1; } },
87            num if num.chars().all(|c| c.is_digit(10)) => number = Some(num.to_string()),
88            _ => {}
89        }
90        i += 1;
91    }
92
93    if do_list_parked { return list_parked(); }
94    if let Some(num) = do_park { return park_episode(&num, &ingest_config); }
95    if let Some((old_num, new_num)) = do_unpark { return unpark_episode(&old_num, &new_num, &ingest_config); }
96
97    if let Some(num) = number {
98        if do_text { 
99            let text_src = ingest_config.text_source.clone().unwrap_or(source.clone());
100            ingest_text(&num, &text_src, title_override, &ingest_config)?; 
101        }
102        if do_image { 
103            let image_src = ingest_config.image_source.clone().unwrap_or(source.clone());
104            ingest_image(&num, &image_src, &ingest_config)?; 
105        }
106        if do_video { 
107            let video_src = ingest_config.video_source.clone().unwrap_or(source.clone());
108            ingest_video(&num, &video_src, &ingest_config)?; 
109        }
110        update_summary(&ingest_config)?;
111    } else {
112        let mut input = String::new();
113        std::io::Read::read_to_string(&mut std::io::stdin(), &mut input)?;
114        let (_ctx, book) = mdbook_preprocessor::parse_input(input.as_bytes())?;
115        let output_json = serde_json::to_string(&book)?;
116        print!("{}", output_json);
117    }
118    Ok(())
119}

121fn ingest_text(number: &str, source: &str, title: Option<&str>, config: &IngestConfig) -> Result<()> {
122    eprintln!("📖 Ingesting text for episode {}...", number);
123    let mut files: Vec<PathBuf> = Vec::new();
124    for ext in &["json", "rs", "md"] {
125        let pattern = format!("{}/*.{}", source, ext);
126        if let Ok(paths) = glob(&pattern) {
127            for path in paths.filter_map(Result::ok) {
128                let filename = path.file_name().unwrap().to_str().unwrap();
129                if !["SUMMARY.md", "cover.md"].contains(&filename) { files.push(path); }
130            }
131        }
132    }
133    files.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
134    if let Some(path) = files.first() {
135        eprintln!("📄 Found source: {}", path.display());
136        let content = std::fs::read_to_string(path)?;
137        let hardened = sanitizer::process_content(content, number, title, config.title_word_limit.unwrap_or(5));
138        std::fs::write(format!("src/{}.md", number), hardened)?;
139        eprintln!("✅ Ingested text to src/{}.md", number);
140    }
141    Ok(())
142}

144fn update_summary(config: &IngestConfig) -> Result<()> {
145    let summary_path = "src/SUMMARY.md";
146    
147    // Pivot and Block Size logic
148    let pivot = 220;
149    let block_size = 21;
150    
151    let mut all_files = Vec::new();
152    let mut wip_parked = Vec::new();
153    
154    let pattern = "src/*.md";
155    for entry in glob(pattern)? {
156        if let Ok(path) = entry {
157            let filename = path.file_name().unwrap().to_str().unwrap();
158            if ["SUMMARY.md", "cover.md", "archive.md", "parked.md", "vault.md", "mempool.md", "github.md", "current.md"].contains(&filename) { continue; }
159            let base = filename.trim_end_matches(".md").trim();
160            let md_content = std::fs::read_to_string(&path)?;
161            let h1_regex = regex::Regex::new(r"(?m)^#\s+(?:(?:\d+)\s*[:\s]*)?\s*(.*)$").unwrap();
162            let title = if let Some(caps) = h1_regex.captures(&md_content) {
163                caps.get(1).unwrap().as_str().trim().to_string()
164            } else { "Untitled".to_string() };
165            let mtime = std::fs::metadata(&path)?.modified()?;
166            let number = base.parse::<u32>().ok();
167            
168            if filename.starts_with('_') {
169                wip_parked.push(EpisodeEntry { number: base.trim_start_matches('_').parse().ok(), filename: base.to_string(), title, mtime });
170            } else {
171                all_files.push(EpisodeEntry { number, filename: base.to_string(), title, mtime });
172            }
173        }
174    }
175
176    // Sort WIP
177    wip_parked.sort_by(|a, b| b.mtime.cmp(&a.mtime));
178
179    // Calculate Blocks
180    // Current Block = Max Block ID
181    // All others = Deep Storage
182    let mut block_map: BTreeMap<i32, Vec<EpisodeEntry>> = BTreeMap::new();
183    let mut thematic_episodes = Vec::new();
184
185    for ep in all_files {
186        if let Some(num) = ep.number {
187            let block_id = ((num as i32 - pivot) / block_size) + 1;
188            block_map.entry(block_id).or_insert_with(Vec::new).push(ep);
189        } else {
190            thematic_episodes.push(ep);
191        }
192    }
193
194    let current_block_id = block_map.keys().last().cloned().unwrap_or(2);
195    let mut blocks: Vec<_> = block_map.into_iter().collect();
196    blocks.sort_by(|a, b| b.0.cmp(&a.0));
197
198    eprintln!("𓄊 Indexer: Current Block ID: {}, Blocks found: {}", current_block_id, blocks.len());
199
200    let original_content = std::fs::read_to_string(summary_path)?;
201    let mut final_lines = Vec::new();
202    
203    for line in original_content.lines() {
204        if line.contains("- [The Mempool") || line.contains("- [Current Block") || line.contains("- [Deep Storage") || line.contains("<!-- RECENT_START -->") { break; }
205        final_lines.push(line.to_string());
206    }
207
208    // 1. Section: The Mempool
209    final_lines.push("\n- [mempool](mempool.md)".to_string());
210    if wip_parked.is_empty() {
211        final_lines.push("    - [None at this moment. Join us on GitHub!](github.md)".to_string());
212    } else {
213        for ep in wip_parked {
214            let display_num = ep.number.map(|n| n.to_string()).unwrap_or_else(|| ep.filename.clone());
215            final_lines.push(format!("    - [{} : {}]({}.md)", display_num, ep.title, ep.filename));
216        }
217    }
218
219    // 2. Section: Current Block
220    final_lines.push("\n<!-- RECENT_START -->".to_string());
221    final_lines.push("- [block template](current.md)".to_string());
222    if let Some((id, eps)) = blocks.iter_mut().find(|(id, _)| *id == current_block_id) {
223        eps.sort_by(|a, b| b.number.unwrap().cmp(&a.number.unwrap()));
224        for ep in eps {
225            final_lines.push(format!("    - [{} : {}]({}.md)", ep.number.unwrap(), ep.title, ep.filename));
226        }
227    }
228    final_lines.push("<!-- RECENT_END -->".to_string());
229
230    // 3. Section: Deep Storage
231    final_lines.push("\n- [chain](archive.md)".to_string());
232    for (id, mut eps) in blocks {
233        if id == current_block_id { continue; }
234        final_lines.push(format!("    - [block {}]()", id));
235        eps.sort_by(|a, b| b.number.unwrap().cmp(&a.number.unwrap()));
236        for ep in eps {
237            final_lines.push(format!("        - [{} : {}]({}.md)", ep.number.unwrap(), ep.title, ep.filename));
238        }
239    }
240
241    // 4. Append Thematic Heritage
242    let mut in_thematic_zone = false;
243    let mut thematic_buffer = Vec::new();
244    let num_regex = regex::Regex::new(r"\d+\.md").unwrap();
245    let skip_strings = ["# WIP", "# Archive", "# Repository", "parked.md", "mempool.md", "Deep Storage", "The Network", "Verified Blocks", "Older Episodes", "github.md", "# Recent Blocks", "# The Mempool", "- [The Mempool", "Current Block", "current.md", "- [The Archive", "Block ", "- [Deep Storage"];
246    
247    for line in original_content.lines() {
248        if line.contains("<!-- RECENT_END -->") { in_thematic_zone = true; continue; }
249        if in_thematic_zone {
250            let mut skip = false;
251            for s in &skip_strings {
252                if line.contains(s) { skip = true; break; }
253            }
254            if skip { continue; }
255            if num_regex.is_match(line) { continue; }
256            
257            // Normalize indentation
258            let mut mod_line = line.to_string();
259            if mod_line.starts_with("  - [") && !mod_line.starts_with("    - [") {
260                mod_line = mod_line.replacen("  - [", "    - [", 1);
261            }
262            if mod_line.starts_with("      - [") && !mod_line.starts_with("        - [") {
263                mod_line = mod_line.replacen("      - [", "        - [", 1);
264            }
265            
266            if thematic_buffer.is_empty() && mod_line.trim().is_empty() { continue; }
267            thematic_buffer.push(mod_line);
268        }
269    }
270    final_lines.extend(thematic_buffer);
271    std::fs::write(summary_path, final_lines.join("\n"))?;
272    
273    // Create core pages
274    let mempool_content = format!("# The Mempool (Unconfirmed Research)\n\nIn a blockchain, the mempool is where transactions wait to be verified. Here, the Mempool contains our raw, unconfirmed ideas. These episodes are currently being researched, debated, and refined. We invite you to act as a validating node—review the research on our GitHub and email your consensus or objections to amj@shutri.com before we mine the next block.\n\n### Offline Access & Contribution\nTo work on these episodes locally, clone the repository:\n\n```bash\ngit clone https://github.com/ashutoshmjain/deepDive.git\n```\n");
275    std::fs::write("src/mempool.md", mempool_content)?;
276
277    let archive_content = format!("# Deep Storage (The Immutable Ledger)\n\nEverything here is organized in Deep Storage. This ledger contains our complete, immutable history. You can expand the folders in the sidebar to browse older **Verified Blocks**, or explore the unnumbered **Genesis Concepts** that built the foundation of our current research framework.\n\n### Collaboration & Offline Access\nFor full offline access to the entire history, or to collaborate on research, please clone our GitHub repository:\n\n```bash\ngit clone https://github.com/ashutoshmjain/deepDive.git\n```\n");
278    std::fs::write("src/archive.md", archive_content)?;
279    
280    let current_content = format!("# Current Block\n\nThis block contains the active research episodes, instantly available for offline reading in the Progressive Web App. Once this block reaches 21 episodes, it is mined and moved to Deep Storage.\n");
281    std::fs::write("src/current.md", current_content)?;
282
283    std::fs::write("src/github.md", "# Join us on GitHub\n\n[Click here to visit the repository](https://github.com/ashutoshmjain/deepDive)")?;
284    Ok(())
285}

287fn ingest_image(number: &str, source: &str, _config: &IngestConfig) -> Result<()> {
288    eprintln!("🎨 Ingesting image for episode {}...", number);
289    let img_dir = "src/img";
290    std::fs::create_dir_all(img_dir)?;
291    let mut images: Vec<PathBuf> = glob(&format!("{}/*{}*.png", source, number))?.filter_map(Result::ok).collect();
292    images.sort_by(|a, b| std::fs::metadata(b).unwrap().modified().unwrap().cmp(&std::fs::metadata(a).unwrap().modified().unwrap()));
293    if let Some(path) = images.first() {
294        let dest = format!("{}/{}.png", img_dir, number);
295        if path.as_path() != std::path::Path::new(&dest) { std::fs::copy(path, &dest)?; }
296    }
297    Ok(())
298}

300fn ingest_video(number: &str, source: &str, _config: &IngestConfig) -> Result<()> {
301    eprintln!("🎬 Ingesting video for episode {}...", number);
302    let vid_dir = "src/vid";
303    std::fs::create_dir_all(vid_dir)?;
304    let pattern = format!("{}/*{}*.mp4", source, number);
305    if let Ok(paths) = glob(&pattern) {
306        for path in paths.filter_map(Result::ok) {
307            let filename = path.file_name().unwrap().to_str().unwrap();
308            std::fs::copy(&path, format!("{}/{}", vid_dir, filename))?;
309        }
310    }
311    Ok(())
312}

319fn list_parked() -> Result<()> {
320    eprintln!("🅿️  Currently Parked Episodes:");
321    for entry in glob("src/_[0-9]*.md")? {
322        if let Ok(path) = entry {
323            let filename = path.file_name().unwrap().to_str().unwrap();
324            let base = filename.trim_end_matches(".md");
325            let content = std::fs::read_to_string(&path)?;
326            let h1_regex = regex::Regex::new(r"(?m)^#\s+(?:(?:\d+)\s*[:\s]*)?\s*(.*)$").unwrap();
327            let title = if let Some(caps) = h1_regex.captures(&content) { caps.get(1).unwrap().as_str().trim() } else { "Untitled" };
328            println!("  [{}] : {}", &base[1..], title);
329        }
330    }
331    Ok(())
332}

334fn park_episode(number: &str, config: &IngestConfig) -> Result<()> {
335    let videos: Vec<_> = glob(&format!("src/vid/{}*.mp4", number))?.filter_map(Result::ok).collect();
336    if !videos.is_empty() { anyhow::bail!("Episode is immutable"); }
337    let src = format!("src/{}.md", number);
338    if std::path::Path::new(&src).exists() {
339        std::fs::rename(&src, format!("src/_{}.md", number))?;
340        update_summary(config)?; 
341    }
342    Ok(())
343}

345fn unpark_episode(old_number: &str, new_number: &str, config: &IngestConfig) -> Result<()> {
346    let src = format!("src/_{}.md", old_number);
347    let dest = format!("src/{}.md", new_number);
348    if !std::path::Path::new(&src).exists() { anyhow::bail!("Source not found"); }
349    std::fs::rename(&src, &dest)?;
350    let mut content = std::fs::read_to_string(&dest)?;
351    content = content.replace(&format!("# {}", old_number), &format!("# {}", new_number));
352    std::fs::write(&dest, content)?;
353    update_summary(config)?;
354    Ok(())
