24/// The primary entry point for the Gemini-to-mdbook transformation.
25pub fn process_content(mut content: String, ep_num: &str, _title_override: Option<&str>, word_limit: usize) -> String {
26    // 0. Decode HTML Entities early
27    content = decode_html_entities(&content).to_string();
28
29    // 0.0 Detect JSON Capsule
30    let mut json_title = None;
31    let mut json_body = None;
32    let mut json_refs = None;
33    let mut is_json_capsule = false;
34
35    // Try parsing as pure JSON first
36    if let Ok(capsule) = serde_json::from_str::<GeminiCapsule>(&content) {
37        json_title = Some(capsule.title);
38        json_body = Some(capsule.body);
39        json_refs = Some(capsule.references);
40        is_json_capsule = true;
41    } else {
42        // Fallback to regex for code block if LLM was "helpful"
43        let json_regex = Regex::new(r"(?s)```json\s*(\{.*?\})\s*```").unwrap();
44        if let Some(caps) = json_regex.captures(&content) {
45            let json_raw = caps.get(1).unwrap().as_str();
46            if let Ok(capsule) = serde_json::from_str::<GeminiCapsule>(json_raw) {
47                json_title = Some(capsule.title);
48                json_body = Some(capsule.body);
49                json_refs = Some(capsule.references);
50                is_json_capsule = true;
51            }
52        }
53    }
54
55    // Prepare content for standard sanitization
56    if is_json_capsule {
57        content = json_body.unwrap();
58    } else {
59        // 0.1 Strip Rust shields (if any) for non-JSON content
60        content = content.trim().to_string();
61        let prefix_regex = Regex::new(r"^(?:Rust)?r(#+)\x22").unwrap();
62        if let Some(caps) = prefix_regex.captures(&content) {
63            let hash_count = caps.get(1).unwrap().as_str().len();
64            content = prefix_regex.replace(&content, "").to_string();
65            let suffix_pattern = format!(r"\x22{}", "#".repeat(hash_count));
66            if content.ends_with(&suffix_pattern) {
67                content = content[..content.len() - suffix_pattern.len()].to_string();
68            } else if content.ends_with(&format!("{}\n", suffix_pattern)) {
69                content = content[..content.len() - suffix_pattern.len() - 1].to_string();
70            }
71        }
72        let gemini_artifacts = Regex::new(r"(?m)^```(text|markdown|rust)?\s*$").unwrap();
73        content = gemini_artifacts.replace_all(&content, "").to_string();
74        content = content.trim_matches('`').trim().to_string();
75    }
76
77    // 1. Title Processing
78    let h1_regex_full = Regex::new(r"(?m)^#\s(?:\d+\s*:\s*)?\s*(.*)$").unwrap();
79    let mut h1_title = if let Some(t) = json_title {
80        t
81    } else if let Some(caps) = h1_regex_full.captures(&content) {
82        caps.get(1).unwrap().as_str().trim().trim_matches('*').to_string()
83    } else {
84        String::from("Untitled")
85    };
86
87    content = h1_regex_full.replace(&content, "").to_string().trim().to_string();
88
89    let words: Vec<&str> = h1_title.split_whitespace().collect();
90    if words.len() > word_limit {
91        h1_title = words[..word_limit].join(" ");
92    }
93
94    // 2. Sanitization Pipeline
95    content = content.replace('\u{0332}', "");
96    let control_chars = Regex::new(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f-\x9f]").unwrap();
97    content = control_chars.replace_all(&content, "").to_string();
98
99    content = convert_ascii_tables(content);
100    content = wrap_ascii_diagrams(content);
101
102    let backslash_cleanup = Regex::new(r"\\([_.\-+!|>\[\]=])").unwrap();
103    content = backslash_cleanup.replace_all(&content, "$1").to_string();
104
105    let math_block_regex = Regex::new(r"(?s)\$\$.*?\$\$").unwrap();
106    let inline_math_regex = Regex::new(r"\$.*?\$").unwrap();
107    let mut math_blocks = Vec::new();
108    let content_with_placeholders = math_block_regex.replace_all(&content, |caps: &regex::Captures| {
109        let placeholder = format!("__MATH_BLOCK_{}__", math_blocks.len());
110        math_blocks.push(caps.get(0).unwrap().as_str().to_string());
111        placeholder
112    }).to_string();
113    let content_with_all_placeholders = inline_math_regex.replace_all(&content_with_placeholders, |caps: &regex::Captures| {
114        let placeholder = format!("__MATH_BLOCK_{}__", math_blocks.len());
115        math_blocks.push(caps.get(0).unwrap().as_str().to_string());
116        placeholder
117    }).to_string();
118
119    let mut temp_content = content_with_all_placeholders;
120
121    // 3. Socials Construction
122    let podcast_links = r#"
123<center><a href="https://open.spotify.com/show/7doWf0GON9JsG6r8igc7RE" target="_blank" style="background-color: #2E2E2E; color: white; padding: 10px 20px; text-align: center; text-decoration: none; display: inline-block; border-radius: 5px; margin-top: 10px; margin-right: 10px;">Spotify</a><a href="https://podcasts.apple.com/us/podcast/deep-dive-with-gemini/id1844532251" target="_blank" style="background-color: #2E2E2E; color: white; padding: 10px 20px; text-align: center; text-decoration: none; display: inline-block; border-radius: 5px; margin-top: 10px; margin-right: 10px;">Apple Podcasts</a><a href="https://fountain.fm/show/7LBvZT6ffpGyubvk8aSF" target="_blank" style="background-color: #2E2E2E; color: white; padding: 10px 20px; text-align: center; text-decoration: none; display: inline-block; border-radius: 5px; margin-top: 10px;">Fountain.fm</a></center>
124"#;
125
126    let lightning_widget = r#"
127<center>
128<lightning-widget
129  name='Thanks for supporting the publication'
130  accent='#f9ce00'
131  to='shutosha@primal.net'
132  image='https://nostrcheck.me/media/5af0794606a15b5641e25aa23d04af4cb0d7d5e68b11cacb47e56a4698fca8c4/49ff6d00cb5bc819cd19f77783d4815fbd46a5b99b6fbdead1eaecfab798187b.webp'
133/>
134</center>
135<script src='https://embed.twentyuno.net/js/app.js'></script>
136"#;
137
138    let social_block = format!("\n\n---\n\n### Tips and Donations\n\nIf you enjoyed this research, consider supporting the project with a tip in **Sats**. It's a simple, global way to support independent research.\n\n<!-- SOCIALS_START -->\n{}{}\n<!-- SOCIALS_END -->\n\nTo send Sats, you'll need a [lightning wallet](https://lightningaddress.com/).\n\n---\n", lightning_widget, podcast_links);
139
140    let mut refs_section = String::from("\n\n#### **Works cited**\n\n");
141    if is_json_capsule {
142        if let Some(refs) = json_refs {
143            let mut markers_found = false;
144            for r in &refs {
145                let marker = format!("[{}]", r.id);
146                if temp_content.contains(&marker) { markers_found = true; break; }
147            }
148
149            for r in refs {
150                let mut ref_text = r.text.clone();
151                let url_regex = Regex::new(r"(https?://[^\s)\]]+[^.\s)\]])").unwrap();
152                ref_text = url_regex.replace_all(&ref_text, |c: &regex::Captures| {
153                    let u = c.get(1).unwrap().as_str();
154                    format!("[{}]({})", u, u)
155                }).to_string();
156
157                if markers_found {
158                    let marker = format!("[{}]", r.id);
159                    let foot_marker = format!("[^{}]", r.id);
160                    temp_content = temp_content.replace(&marker, &foot_marker);
161                    refs_section.push_str(&format!("{}: {}\n\n", foot_marker, ref_text));
162                } else {
163                    refs_section.push_str(&format!("{}. {}\n\n", r.id, ref_text));
164                }
165            }
166        }
167    } else {
168        temp_content = fix_footnotes(temp_content, &social_block);
169        refs_section = String::new(); 
170    }
171
172    // 4. Currency and Dollar Escaping
173    let currency_regex = Regex::new(r"(?i)\$([\d\.,]+)\s*(million|billion|trillion|k|m|b|t)?").unwrap();
174    temp_content = currency_regex.replace_all(&temp_content, "$1 $2 USD ").to_string();
175    temp_content = temp_content.replace("  ", " ");
176    temp_content = temp_content.replace("$", r"\$");
177
178    // 5. Restore Math Blocks
179    for (idx, block) in math_blocks.iter().enumerate() {
180        let placeholder = format!("__MATH_BLOCK_{}__", idx);
181        temp_content = temp_content.replace(&placeholder, block);
182    }
183
184    // 6. Final Assembly
185    if is_json_capsule {
186        format!("# {} : {}\n\n{}{}{}{}", ep_num, h1_title, temp_content.trim(), social_block, refs_section, "\n").trim().to_string()
187    } else {
188        // For non-JSON (like our payload), fix_footnotes now handles the merge correctly
189        format!("# {} : {}\n\n{}", ep_num, h1_title, temp_content.trim()).trim().to_string()
190    }
191}

193pub fn find_first_substantial_paragraph(content: &str) -> Option<usize> {
194    let mut in_code_block = false;
195    let mut current_pos = 0;
196    for line in content.lines() {
197        let trimmed = line.trim();
198        if trimmed.starts_with("```") {
199            in_code_block = !in_code_block;
200            current_pos += line.len() + 1;
201            continue;
202        }
203        if !in_code_block && !trimmed.is_empty() {
204            let is_header = trimmed.starts_with('#');
205            let is_list = trimmed.starts_with('*') || trimmed.starts_with('-') || (trimmed.len() > 2 && trimmed.chars().next().unwrap().is_digit(10) && trimmed.contains(". "));
206            let is_html = trimmed.starts_with('<');
207            if !is_header && !is_list && !is_html && trimmed.len() > 100 {
208                return Some(current_pos + line.len());
209            }
210        }
211        current_pos += line.len() + 1;
212    }
213    None
214}

216fn wrap_ascii_diagrams(content: String) -> String {
217    let mut result = String::new();
218    let lines: Vec<&str> = content.lines().collect();
219    let mut i = 0;
220    
221    let is_diag_line = |s: &str| -> bool {
222        let t = s.trim();
223        if t.is_empty() { return false; }
224        if t == "v" || t == "|" || t == "^" || t == "v |" { return true; }
225        if t.starts_with("v ") || t.starts_with("| ") { return true; }
226        if t.contains("===>") || t.contains("<===") || t.contains("<====") { return true; }
227        if t.starts_with('[') && t.contains(']') && !t.contains("](") { 
228            let citation_regex = Regex::new(r"^\[\d+\]").unwrap();
229            if citation_regex.is_match(t) { return false; }
230            return true; 
231        }
232        if t.starts_with("**<---") || t.starts_with("--->**") { return true; }
233        if t.starts_with("**") && (t.ends_with("v**") || t.ends_with("|**")) { return true; }
234        if t.starts_with("+---") && t.ends_with("+") { return true; }
235        if t.contains("|") && (t.contains("+--->") || t.contains("ALGEBRAIC ASCENT")) { return true; }
236        if t.starts_with("=====") { return true; }
237        if t.contains("<====") { return true; }
238        if t.contains("+---") && t.contains("+") { return true; }
239        false
240    };
241
242    while i < lines.len() {
243        if is_diag_line(lines[i]) {
244            let mut last_diag_idx = i;
245            let mut j = i + 1;
246            while j < lines.len() {
247                let t = lines[j].trim();
248                if is_diag_line(lines[j]) {
249                    last_diag_idx = j;
250                } else if !t.is_empty() {
251                    let is_indented = lines[j].starts_with("  ");
252                    let mut next_is_diag = false;
253                    for k in 1..=3 {
254                        if j + k < lines.len() && is_diag_line(lines[j + k]) {
255                            next_is_diag = true;
256                            break;
257                        }
258                    }
259                    if !is_indented || !next_is_diag { break; }
260                }
261                j += 1;
262            }
263            let is_multi_line = last_diag_idx > i;
264            let contains_arrows = lines[i].contains("===>") || lines[i].contains("<===") || lines[i].contains("<====") || lines[i].starts_with("=====");
265            let has_table_separator = (i..=last_diag_idx).any(|k| lines[k].contains("--- |") || lines[k].contains("| ---") || (lines[k].contains('|') && lines[k].contains('-')));
266            
267            if (is_multi_line || contains_arrows) && !has_table_separator {
268                result.push_str("\n```text\n");
269                for k in i..=last_diag_idx {
270                    result.push_str(lines[k]);
271                    result.push('\n');
272                }
273                result.push_str("```\n");
274            } else {
275                for k in i..=last_diag_idx {
276                    result.push_str(lines[k]);
277                    result.push('\n');
278                }
279            }
280            i = last_diag_idx + 1;
281        } else {
282            result.push_str(lines[i]);
283            result.push('\n');
284            i += 1;
285        }
286    }
287    result
288}

290fn fix_footnotes(content: String, social_block: &str) -> String {
291    let header_regex = Regex::new(r"(?im)^#+\s+(\*\*Works cited\*\*|Works cited|References|Bibliography|References or Bibliography)").unwrap();
292    let parts: Vec<&str> = header_regex.split(&content).collect();
293    
294    // 1. Standardize Body Markers
295    let body_raw = parts[0];
296    
297    // Transform [N] or [^N] or [N, M] to standard [N][M] markers
298    // We strip the ^ if it exists to normalize to [N] for the internal index
299    let marker_regex = Regex::new(r"\[\^?(\d+(?:\s*[,\s]\s*\d+)*)\]").unwrap();
300    let body_normalized = marker_regex.replace_all(body_raw, |caps: &regex::Captures| {
301        let content = caps.get(1).unwrap().as_str();
302        let nums = re_split_nums(content);
303        nums.iter().map(|n| format!("[{}]", n)).collect::<Vec<_>>().join("")
304    }).to_string();
305
306    if parts.len() < 2 { 
307        let mut res = body_normalized.trim().to_string();
308        res.push_str(social_block);
309        return res;
310    }
311
312    let refs_raw = parts[1];
313    let header = "#### **Works cited**";
314
315    // 2. Parse Bibliography
316    // Matches: "[N]: text" or "[^N]: text" or "N. text"
317    let ref_pattern = Regex::new(r"(?m)^(\*?\s*(\[?\^?(\d+)\]?[:\.]?)\s*|\s*$)").unwrap();
318    let mut ref_entries = Vec::new();
319    let matches: Vec<_> = ref_pattern.find_iter(refs_raw).collect();
320    for (i, m) in matches.iter().enumerate() {
321        let caps = ref_pattern.captures(m.as_str()).unwrap();
322        let old_num = caps.get(3).map(|n| n.as_str().to_string());
323        let start = m.end();
324        let end = if i + 1 < matches.len() { matches[i+1].start() } else { refs_raw.len() };
325        let text = refs_raw[start..end].trim().to_string();
326        if let Some(num) = old_num {
327            if !text.is_empty() {
328                ref_entries.push(RefEntry { old_num: Some(num), text, processed: false });
329            }
330        }
331    }
332
333    // 3. Build Sequential Index
334    let mut old_to_new = HashMap::new();
335    let mut new_refs = Vec::new();
336    let url_regex = Regex::new(r"(https?://[^\s)\]]+[^.\s)\]])").unwrap();
337    
338    // Find all [N] markers in body and map to 1, 2, 3...
339    let norm_marker_regex = Regex::new(r"\[(\d+)\]").unwrap();
340    let mut unique_old_nums = Vec::new();
341    for caps in norm_marker_regex.captures_iter(&body_normalized) {
342        let n = caps.get(1).unwrap().as_str().to_string();
343        if !unique_old_nums.contains(&n) { unique_old_nums.push(n); }
344    }
345
346    for old_num in unique_old_nums {
347        let mut aggregated_text = String::new();
348        let mut found = false;
349        // Collect all text pieces for this citation number
350        for ref_entry in ref_entries.iter_mut().filter(|r| r.old_num.as_ref() == Some(&old_num)) {
351            if found { aggregated_text.push_str(" "); }
352            let entry_text_clean = ref_entry.text.replace("`", "");
353            let processed_text = url_regex.replace_all(&entry_text_clean, |caps: &regex::Captures| {
354                let url = caps.get(1).unwrap().as_str();
355                format!("[{}]({})", url, url)
356            }).to_string();
357            aggregated_text.push_str(&processed_text);
358            ref_entry.processed = true;
359            found = true;
360        }
361        
362        let new_num = (new_refs.len() + 1).to_string();
363        old_to_new.insert(old_num.clone(), new_num);
364        if found {
365            new_refs.push(aggregated_text);
366        } else {
367            new_refs.push(format!("**TODO: Missing citation for index {}**", old_num));
368        }
369    }
370
371    // 4. Final Body and Footer Assembly
372    let final_body = norm_marker_regex.replace_all(&body_normalized, |caps: &regex::Captures| {
373        let n = caps.get(1).unwrap().as_str();
374        format!("[^{}] ", old_to_new.get(n).unwrap_or(&n.to_string())) // Added trailing space
375    }).to_string().replace(" ]", "]"); // Clean up trailing space if inside brackets (though markers are outside)
376
377    let mut result = final_body.to_string();
378    result.push_str(social_block);
379    result.push_str("\n\n");
380    result.push_str(header);
381    result.push_str("\n\n");
382    for (i, text) in new_refs.iter().enumerate() {
383        result.push_str(&format!("[^{}]: {}\n\n", i + 1, text));
384    }
385    result
386}

389fn re_split_nums(s: &str) -> Vec<String> {
390    let re = Regex::new(r"[,\s]+").unwrap();
391    re.split(s).map(|n| n.trim().to_string()).filter(|n| !n.is_empty()).collect()
392}

403fn convert_ascii_tables(content: String) -> String {
404    let mut result = String::new();
405    let mut lines = content.lines().peekable();
406    while let Some(line) = lines.next() {
407        let trimmed = line.trim();
408        let is_table_start = (trimmed.contains("+---") && trimmed.ends_with('+')) || (trimmed.starts_with('|') && (trimmed.ends_with('|') || trimmed.contains(" --- ")));
409        if is_table_start {
410            let mut table_data = Vec::new();
411            table_data.push(line);
412            while let Some(next) = lines.peek() {
413                let nt = next.trim();
414                if nt.starts_with('|') || (nt.contains("+---") && nt.ends_with('+')) || nt.contains(" --- ") || nt.is_empty() {
415                    table_data.push(lines.next().unwrap());
416                } else { break; }
417            }
418            let mut md_rows = Vec::new();
419            for row in table_data {
420                let r = row.trim();
421                if r.starts_with('|') {
422                    let cells: Vec<String> = r.split('|').map(|s| s.trim().to_string()).filter(|s| !s.is_empty()).collect();
423                    if !cells.is_empty() { md_rows.push(cells); }
424                }
425            }
426            if !md_rows.is_empty() {
427                result.push('\n');
428                let mut start_idx = 0;
429                if md_rows[0].len() == 1 && md_rows.len() > 1 {
430                    result.push_str(&format!("**{}**\n\n", md_rows[0][0]));
431                    start_idx = 1;
432                }
433                if md_rows.len() > start_idx {
434                    for (i, row) in md_rows[start_idx..].iter().enumerate() {
435                        result.push_str(&format!("| {} |\n", row.join(" | ")));
436                        if i == 0 {
437                            let sep: Vec<String> = row.iter().map(|_| "---".to_string()).collect();
438                            result.push_str(&format!("| {} |\n", sep.join(" | ")));
439                        }
440                    }
441                }
442                result.push('\n');
443            }
444        } else {
445            result.push_str(line);
446            result.push('\n');
447        }
448    }
449    result
450}
