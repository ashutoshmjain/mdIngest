1//! # Sanitizer Module (Gemini-to-mdbook)
2//! 
3//! This module provides the core transformation logic for converting 
4//! "shielded" Gemini Pro outputs into production-ready mdbook content.
5//! 
6//! It is specifically designed to handle the constraints of the **Master Prompt**, 
7//! including the removal of Rust-style raw string literals (`r#" ... "#`) and 
8//! the enforcement of research-tier formatting.
9
10use regex::Regex;
11use std::collections::HashMap;
12use html_escape::decode_html_entities;
13use serde::Deserialize;
14
15#[derive(Debug, Deserialize)]
16struct GeminiCapsule {
17    title: String,
18    body: String,
19    references: Vec<GeminiReference>,
20}
21
22#[derive(Debug, Deserialize)]
23struct GeminiReference {
24    id: usize,
25    text: String,
26}

28/// The primary entry point for the Gemini-to-mdbook transformation.
29pub fn process_content(mut content: String, ep_num: &str, title_override: Option<&str>, word_limit: usize) -> String {
30    // 0. Decode HTML Entities early
31    content = decode_html_entities(&content).to_string();
32
33    // 0.0 Detect JSON Capsule (Pure JSON or JSON in code block)
34    let mut title = None;
35    let mut body = None;
36    let mut references = None;
37
38    // Try parsing as pure JSON first
39    if let Ok(capsule) = serde_json::from_str::<GeminiCapsule>(&content) {
40        title = Some(capsule.title);
41        body = Some(capsule.body);
42        references = Some(capsule.references);
43    } else {
44        // Fallback to regex for code block
45        let json_regex = Regex::new(r"(?s)```json\s*(\{.*?\})\s*```").unwrap();
46        if let Some(caps) = json_regex.captures(&content) {
47            let json_raw = caps.get(1).unwrap().as_str();
48            if let Ok(capsule) = serde_json::from_str::<GeminiCapsule>(json_raw) {
49                title = Some(capsule.title);
50                body = Some(capsule.body);
51                references = Some(capsule.references);
52            }
53        }
54    }
55
56    if let (Some(t), Some(mut b), Some(refs)) = (title, body, references) {
57        let mut refs_section = String::from("\n\n#### **Works cited**\n\n");
58        for r in refs {
59            let marker = format!("[{}]", r.id);
60            let foot_marker = format!("[^{}]", r.id);
61            b = b.replace(&marker, &foot_marker);
62            
63            let mut ref_text = r.text.clone();
64            let url_regex = Regex::new(r"(https?://[^\s)\]]+[^.\s)\]])").unwrap();
65            ref_text = url_regex.replace_all(&ref_text, |c: &regex::Captures| {
66                let u = c.get(1).unwrap().as_str();
67                format!("[{}]({})", u, u)
68            }).to_string();
69            
70            refs_section.push_str(&format!("{}: {}\n\n", foot_marker, ref_text));
71        }
72        content = format!("# {}\n\n{}{}", t, b, refs_section);
73    }
74
75    // 0.1 Strip Rust raw string artifacts and Gemini markdown markers (The "Shield")
76    content = content.trim().to_string();
77    
78    // Handle variable number of hashes in raw string: r#", r##", r###", etc.
79    let prefix_regex = Regex::new(r"^(?:Rust)?r(#+)\x22").unwrap();
80    let mut hash_count = 0;
81    
82    if let Some(caps) = prefix_regex.captures(&content) {
83        hash_count = caps.get(1).unwrap().as_str().len();
84        content = prefix_regex.replace(&content, "").to_string();
85    }
86
87    if hash_count > 0 {
88        let suffix_pattern = format!(r"\x22{}", "#".repeat(hash_count));
89        if content.ends_with(&suffix_pattern) {
90            content = content[..content.len() - suffix_pattern.len()].to_string();
91        } else if content.ends_with(&format!("{}\n", suffix_pattern)) {
92            content = content[..content.len() - suffix_pattern.len() - 1].to_string();
93        }
94    }
95
96    let gemini_artifacts = Regex::new(r"(?m)^```(text|markdown|rust)?\s*$").unwrap();
97    content = gemini_artifacts.replace_all(&content, "").to_string();
98    
99    // Final trim to handle any leftover backticks at top/bottom
100    content = content.trim_matches('`').trim().to_string();
101
102    // 0.2 Strip orphaned base64 images early to prevent interference with diagram wrapping
103    let image_def_regex = Regex::new(r"(?m)^\[image\d+\]: <data:image/.*?>\s*$").unwrap();
104    content = image_def_regex.replace_all(&content, "").to_string();
105
106    // [NEW] Socials & Monetization Injection (Immediately after H1)
107    let podcast_links = r#"
108<center><a href="https://open.spotify.com/show/7doWf0GON9JsG6r8igc7RE" target="_blank" style="background-color: #2E2E2E; color: white; padding: 10px 20px; text-align: center; text-decoration: none; display: inline-block; border-radius: 5px; margin-top: 10px; margin-right: 10px;">Spotify</a><a href="https://podcasts.apple.com/us/podcast/deep-dive-with-gemini/id1844532251" target="_blank" style="background-color: #2E2E2E; color: white; padding: 10px 20px; text-align: center; text-decoration: none; display: inline-block; border-radius: 5px; margin-top: 10px; margin-right: 10px;">Apple Podcasts</a><a href="https://fountain.fm/show/7LBvZT6ffpGyubvk8aSF" target="_blank" style="background-color: #2E2E2E; color: white; padding: 10px 20px; text-align: center; text-decoration: none; display: inline-block; border-radius: 5px; margin-top: 10px;">Fountain.fm</a></center>
109"#;
110
111    let lightning_widget = r#"
112<center><a href="lightning:shutosha@primal.net" style="background-color: #F7931A; color: white; padding: 10px 20px; text-align: center; text-decoration: none; display: inline-block; border-radius: 5px; margin-top: 10px; font-weight: bold;">⚡ Zap with Lightning</a></center>
113"#;
114
115    let injection_block = format!("\n<!-- SOCIALS_START -->\n{}{}\n<!-- SOCIALS_END -->\n", podcast_links, lightning_widget);
116
117    // 1. Title Sync & Legacy Removal
118    let h1_regex_full = Regex::new(r"(?m)^#\s(?:\d+\s*:\s*)?\s*(.*)$").unwrap();
119    let mut h1_title = String::from("Untitled");
120
121    if let Some(caps) = h1_regex_full.captures(&content) {
122        h1_title = caps.get(1).unwrap().as_str().trim().trim_matches('*').to_string();
123        content = h1_regex_full.replace(&content, "").to_string();
124    }
125
126    // Standardize title word limit
127    let words: Vec<&str> = h1_title.split_whitespace().collect();
128    if words.len() > word_limit {
129        h1_title = words[..word_limit].join(" ");
130    }
131    
132    // Reconstruct with Injection Block
133    content = format!("# {} : {}{}{}", ep_num, h1_title, injection_block, content.trim());
134
135    // 2. Invisible Character Sanitization
136    content = content.replace('\u{0332}', "");
137    let control_chars = Regex::new(r"[\x00-\x08\x0b\x0c\x0e-\x1f\x7f-\x9f]").unwrap();
138    content = control_chars.replace_all(&content, "").to_string();
139
140    // 3. Convert ASCII Tables to Markdown Tables (DO THIS BEFORE DIAGRAMS)
141    content = convert_ascii_tables(content);
142
143    // 4. Wrap ASCII Diagrams in code blocks
144    content = wrap_ascii_diagrams(content);
145
146    // 5. Backslash cleanup (Surgical - only common AI escapes)
147    let backslash_cleanup = Regex::new(r"\\([_.\-+!|>\[\]=])").unwrap();
148    content = backslash_cleanup.replace_all(&content, "$1").to_string();
149
150    // 6. Preserve Math Blocks
151    let math_block_regex = Regex::new(r"(?s)\$\$.*?\$\$").unwrap();
152    let inline_math_regex = Regex::new(r"\$.*?\$").unwrap();
153    let mut math_blocks = Vec::new();
154    let content_with_placeholders = math_block_regex.replace_all(&content, |caps: &regex::Captures| {
155        let placeholder = format!("__MATH_BLOCK_{}__", math_blocks.len());
156        math_blocks.push(caps.get(0).unwrap().as_str().to_string());
157        placeholder
158    }).to_string();
159    let content_with_all_placeholders = inline_math_regex.replace_all(&content_with_placeholders, |caps: &regex::Captures| {
160        let placeholder = format!("__MATH_BLOCK_{}__", math_blocks.len());
161        math_blocks.push(caps.get(0).unwrap().as_str().to_string());
162        placeholder
163    }).to_string();
164
165    let mut temp_content = content_with_all_placeholders;
166
167    // 7. Fix Footnotes
168    temp_content = fix_footnotes(temp_content);
169
170    // 8. Currency and Dollar Escaping
171    let currency_regex = Regex::new(r"(?i)\$([\d\.,]+)\s*(million|billion|trillion|k|m|b|t)?").unwrap();
172    temp_content = currency_regex.replace_all(&temp_content, "$1 $2 USD ").to_string();
173    temp_content = temp_content.replace("  ", " ");
174    temp_content = temp_content.replace("$", r"\$");
175
176    // 9. Clean any remaining image tags if they exist
177    let image_tag_regex = Regex::new(r"!\[\]\[image\d+\]").unwrap();
178    temp_content = image_tag_regex.replace_all(&temp_content, "").to_string();
179
180    // 10. Restore Math Blocks
181    for (idx, block) in math_blocks.iter().enumerate() {
182        let placeholder = format!("__MATH_BLOCK_{}__", idx);
183        temp_content = temp_content.replace(&placeholder, block);
184    }
185
186    temp_content.trim().to_string()
187}

189/// Identifies the first "substantial" paragraph for smart anchor placement.
190/// 
191/// A substantial paragraph is:
192/// - Not a header (#)
193/// - Not a diagram/code block (```)
194/// - Not a list item (*, -, 1.)
195/// - Not an HTML block (<)
196/// - At least 100 characters long.
197pub fn find_first_substantial_paragraph(content: &str) -> Option<usize> {
198    let mut in_code_block = false;
199    let mut current_pos = 0;
200
201    for line in content.lines() {
202        let trimmed = line.trim();
203
204        if trimmed.starts_with("```") {
205            in_code_block = !in_code_block;
206            current_pos += line.len() + 1;
207            continue;
208        }
209
210        if !in_code_block && !trimmed.is_empty() {
211            let is_header = trimmed.starts_with('#');
212            let is_list = trimmed.starts_with('*') || trimmed.starts_with('-') || (trimmed.len() > 2 && trimmed.chars().next().unwrap().is_digit(10) && trimmed.contains(". "));
213            let is_html = trimmed.starts_with('<');
214
215            if !is_header && !is_list && !is_html && trimmed.len() > 100 {
216                return Some(current_pos + line.len());
217            }
218        }
219        current_pos += line.len() + 1;
220    }
221    None
222}

224/// Identifies and wraps ASCII-based diagrams in code blocks.
225/// 
226/// Logic detects common diagram markers like `===>`, `|`, and `v` while 
227/// ignoring standard citations or Markdown lists.
228fn wrap_ascii_diagrams(content: String) -> String {
229    let mut result = String::new();
230    let lines: Vec<&str> = content.lines().collect();
231    let mut i = 0;
232    
233    let is_diag_line = |s: &str| -> bool {
234        let t = s.trim();
235        if t.is_empty() { return false; }
236        if t == "v" || t == "|" || t == "^" || t == "v |" { return true; }
237        if t.starts_with("v ") || t.starts_with("| ") { return true; }
238        if t.contains("===>") || t.contains("<===") || t.contains("<====") { return true; }
239        if t.starts_with('[') && t.contains(']') && !t.contains("](") { 
240            let citation_regex = Regex::new(r"^\[\d+\]").unwrap();
241            if citation_regex.is_match(t) { return false; }
242            return true; 
243        }
244        if t.starts_with("**<---") || t.starts_with("--->**") { return true; }
245        if t.starts_with("**") && (t.ends_with("v**") || t.ends_with("|**")) { return true; }
246        if t.starts_with("+---") && t.ends_with("+") { return true; }
247        if t.contains("|") && (t.contains("+--->") || t.contains("ALGEBRAIC ASCENT")) { return true; }
248        if t.starts_with("=====") { return true; }
249        if t.contains("<====") { return true; }
250        if t.contains("+---") && t.contains("+") { return true; }
251        false
252    };
253
254    while i < lines.len() {
255        if is_diag_line(lines[i]) {
256            let mut last_diag_idx = i;
257            let mut j = i + 1;
258            
259            while j < lines.len() {
260                let t = lines[j].trim();
261                if is_diag_line(lines[j]) {
262                    last_diag_idx = j;
263                } else if !t.is_empty() {
264                    let is_indented = lines[j].starts_with("  ");
265                    let mut next_is_diag = false;
266                    for k in 1..=3 {
267                        if j + k < lines.len() && is_diag_line(lines[j + k]) {
268                            next_is_diag = true;
269                            break;
270                        }
271                    }
272                    if !is_indented || !next_is_diag {
273                        break;
274                    }
275                }
276                j += 1;
277            }
278            
279            let is_multi_line = last_diag_idx > i;
280            let contains_arrows = lines[i].contains("===>") || lines[i].contains("<===") || lines[i].contains("<====") || lines[i].starts_with("=====");
281            let has_table_separator = (i..=last_diag_idx).any(|k| lines[k].contains("--- |") || lines[k].contains("| ---"));
282            
283            if (is_multi_line || contains_arrows) && !has_table_separator {
284                result.push_str("\n```text\n");
285                for k in i..=last_diag_idx {
286                    result.push_str(lines[k]);
287                    result.push('\n');
288                }
289                result.push_str("```\n");
290            } else {
291                for k in i..=last_diag_idx {
292                    result.push_str(lines[k]);
293                    result.push('\n');
294                }
295            }
296            
297            i = last_diag_idx + 1;
298        } else {
299            result.push_str(lines[i]);
300            result.push('\n');
301            i += 1;
302        }
303    }
304    result
305}

307/// Standardizes and re-indexes footnotes.
308fn fix_footnotes(content: String) -> String {
309    let header_regex = Regex::new(r"(?im)^#+\s+(\*\*Works cited\*\*|Works cited|References|Bibliography|References or Bibliography)").unwrap();
310    let parts: Vec<&str> = header_regex.split(&content).collect();
311    if parts.len() < 2 { return content; }
312
313    let body = parts[0];
314    let refs_raw = parts[1];
315    let header = "#### **Works cited**";
316
317    // Detect entries starting with [number], **[number]**, or even just the text
318    let ref_pattern = Regex::new(r"(?m)^(\*?\s*(\*\*\[?(\d+)\]?\*\*|\[(\d+)\])\s*|\s*$)").unwrap();
319    
320    let mut ref_entries = Vec::new();
321    let matches: Vec<_> = ref_pattern.find_iter(refs_raw).collect();
322    
323    for (i, m) in matches.iter().enumerate() {
324        let caps = ref_pattern.captures(m.as_str()).unwrap();
325        let old_num = caps.get(2).map(|n| n.as_str().to_string())
326            .or_else(|| caps.get(3).map(|n| n.as_str().to_string()));
327        
328        let start = m.end();
329        let end = if i + 1 < matches.len() {
330            matches[i+1].start()
331        } else {
332            refs_raw.len()
333        };
334        
335        let text = refs_raw[start..end].trim().to_string();
336        ref_entries.push(RefEntry { old_num, text, processed: false });
337    }
338
339    if ref_entries.is_empty() { return content; }
340
341    let marker_pattern = Regex::new(r"\[(\d+(?:\s*,\s*\d+)*)\]").unwrap();
342    let mut unique_old_nums = Vec::new();
343    
344    for caps in marker_pattern.captures_iter(body) {
345        let nums_str = caps.get(1).unwrap().as_str();
346        for n in nums_str.split(',') {
347            let n_trimmed = n.trim().to_string();
348            if !unique_old_nums.contains(&n_trimmed) {
349                unique_old_nums.push(n_trimmed);
350            }
351        }
352    }
353
354    let mut old_to_new = HashMap::new();
355    let mut new_refs = Vec::new();
356    let url_regex = Regex::new(r"(https?://[^\s)\]]+[^.\s)\]])").unwrap();
357    
358    for old_num in unique_old_nums {
359        let mut aggregated_text = String::new();
360        let mut found = false;
361        
362        while let Some(ref_entry) = ref_entries.iter_mut().find(|r| r.old_num.as_ref() == Some(&old_num) && !r.processed) {
363            if found { aggregated_text.push_str("\n\n"); }
364            
365            let entry_text_clean = ref_entry.text.replace("`", "");
366            let processed_text = url_regex.replace_all(&entry_text_clean, |caps: &regex::Captures| {
367                let url = caps.get(1).unwrap().as_str();
368                format!("[{}]({})", url, url)
369            }).to_string();
370            
371            aggregated_text.push_str(&processed_text);
372            ref_entry.processed = true;
373            found = true;
374        }
375
376        if found {
377            let new_num = (new_refs.len() + 1).to_string();
378            old_to_new.insert(old_num, new_num);
379            new_refs.push(aggregated_text);
380        } else if let Some(unprocessed_star) = ref_entries.iter_mut().find(|r| r.old_num.is_none() && !r.processed) {
381            let new_num = (new_refs.len() + 1).to_string();
382            old_to_new.insert(old_num, new_num);
383            
384            let processed_text = url_regex.replace_all(&unprocessed_star.text, |caps: &regex::Captures| {
385                let url = caps.get(1).unwrap().as_str();
386                format!("[{}]({})", url, url)
387            }).to_string();
388            
389            new_refs.push(processed_text);
390            unprocessed_star.processed = true;
391        } else {
392            let new_num = (new_refs.len() + 1).to_string();
393            old_to_new.insert(old_num.clone(), new_num);
394            new_refs.push(format!("**TODO: Missing citation for index {}**", old_num));
395        }
396    }
397
398    let final_body = marker_pattern.replace_all(body, |caps: &regex::Captures| {
399        let nums_str = caps.get(1).unwrap().as_str();
400        let mut new_markers: Vec<String> = nums_str.split(',')
401            .map(|n| {
402                let n_trimmed = n.trim();
403                format!("[^{}]", old_to_new.get(n_trimmed).unwrap_or(&n_trimmed.to_string()))
404            }).collect();
405        new_markers.dedup();
406        new_markers.join(" ") 
407    });
408
409    let mut result = final_body.to_string();
410    result.push_str("\n\n");
411    result.push_str(header);
412    result.push_str("\n\n");
413    
414    for (i, text) in new_refs.iter().enumerate() {
415        result.push_str(&format!("[^{}]: {}\n\n", i + 1, text));
416    }
417    result
418}

420struct RefEntry {
421    old_num: Option<String>,
422    text: String,
423    processed: bool,
424}

426/// Identifies and converts ASCII grid tables to standard Markdown tables.
427fn convert_ascii_tables(content: String) -> String {
428    let mut result = String::new();
429    let mut lines = content.lines().peekable();
430
431    while let Some(line) = lines.next() {
432        let trimmed = line.trim();
433        let is_table_start = (trimmed.contains("+---") && trimmed.ends_with('+')) || 
434                             (trimmed.starts_with('|') && trimmed.ends_with('|')) ||
435                             (trimmed.starts_with("![][image") && trimmed.contains("+---"));
436        
437        if is_table_start {
438            let mut table_data = Vec::new();
439            table_data.push(line);
440            while let Some(next) = lines.peek() {
441                let nt = next.trim();
442                if nt.starts_with('|') || (nt.contains("+---") && nt.ends_with('+')) || nt.is_empty() {
443                    table_data.push(lines.next().unwrap());
444                } else {
445                    break;
446                }
447            }
448            let mut md_rows = Vec::new();
449            for row in table_data {
450                let mut r = row.trim().to_string();
451                if r.starts_with("![][image") {
452                    if let Some(pos) = r.find('+') { r = r[pos..].to_string(); }
453                    else if let Some(pos) = r.find('|') { r = r[pos..].to_string(); }
454                }
455                if r.starts_with('|') {
456                    let cells: Vec<String> = r.split('|').map(|s| s.trim().to_string()).filter(|s| !s.is_empty()).collect();
457                    if !cells.is_empty() { md_rows.push(cells); }
458                }
459            }
460            if !md_rows.is_empty() {
461                let is_diagram = md_rows.iter().any(|row| row.iter().any(|cell| cell.contains("ALGEBRAIC ASCENT") || cell.contains("+--->")));
462                if is_diagram {
463                    for row in md_rows {
464                        result.push_str(&format!("| {} |\n", row.join(" | ")));
465                    }
466                    continue;
467                }
468
469                result.push('\n');
470                let mut start_idx = 0;
471                if md_rows[0].len() == 1 {
472                    result.push_str(&format!("**{}**\n\n", md_rows[0][0]));
473                    start_idx = 1;
474                }
475                if md_rows.len() > start_idx {
476                    for (i, row) in md_rows[start_idx..].iter().enumerate() {
477                        result.push_str(&format!("| {} |\n", row.join(" | ")));
478                        if i == 0 {
479                            let sep: Vec<String> = row.iter().map(|_| "---".to_string()).collect();
480                            result.push_str(&format!("| {} |\n", sep.join(" | ")));
481                        }
482                    }
483                }
484                result.push('\n');
485            }
486        } else {
487            result.push_str(line);
488            result.push('\n');
489        }
490    }
491    result
492}
